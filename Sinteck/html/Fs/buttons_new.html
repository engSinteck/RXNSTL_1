document.addEventListener('DOMContentLoaded', () => { // Seleciona todos os controles de spinner const spinnerControls = document.querySelectorAll('.spinner-control'); spinnerControls.forEach(control => { const minusBtn = control.querySelector('.minus-btn'); const plusBtn = control.querySelector('.plus-btn'); const display = control.querySelector('.value-display'); // Define o passo (step) e valor inicial // Usamos um passo de 0.1 para 'RDS level' e 1 para 'Maximum Frequency' const isFloat = display.getAttribute('data-value').includes('.'); const step = isFloat ? 0.1 : 1; minusBtn.addEventListener('click', () => { let currentValue = parseFloat(display.getAttribute('data-value')); currentValue = currentValue - step; // Corrige problemas de precisão do ponto flutuante no JS const newValue = Math.round(currentValue * (1 / step)) / (1 / step); display.value = isFloat ? newValue.toFixed(1) : newValue; // Formata para 1 casa decimal se for float display.setAttribute('data-value', newValue); }); plusBtn.addEventListener('click', () => { let currentValue = parseFloat(display.getAttribute('data-value')); currentValue = currentValue + step; const newValue = Math.round(currentValue * (1 / step)) / (1 / step); display.value = isFloat ? newValue.toFixed(1) : newValue; display.setAttribute('data-value', newValue); }); }); });
